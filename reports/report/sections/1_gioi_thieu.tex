\section{Giới thiệu và mô tả bài toán}
\phutrach{A (trình bày phần bài toán, dữ liệu)}

\subsection{Bối cảnh thực tế}
\canviet{phát nhạc liên tục (tự phát tiếp, nhạc nền quán, tiệc, tập gym): bài kế tiếp hợp gu nhưng lệch nhịp, lệch tông, tụt năng lượng; ví dụ một lần chuyển bài hỏng}

\subsection{Phát biểu bài toán}
\canviet{đầu vào: phiên (seed 5 bài); đầu ra: top-10 bài tiếp theo; hai mục tiêu: đúng gu (bài đích, NDCG@10) và mix mượt (điểm mix, tỉ lệ cặp mix được); $\alpha$ cân giữa hai mục tiêu}

\subsection{Đóng góp của nhóm}
\canviet{tóm tắt 3--4 ý: hệ hai tầng, điểm mix theo luật và học, đường đánh đổi, demo}
